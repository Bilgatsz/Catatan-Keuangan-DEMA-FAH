import pandas as pd import streamlit as st # Konfigurasi halaman agar pas di layar iPhone st.set_page_config( page_title="Keuangan BEM", page_icon="💰", layout="centered" ) st.title("📱 Keuangan BEM") st.write("Catat pemasukan dan pengeluaran dengan mudah dari iPhone kamu.") # 1. Inisialisasi Database Sederhana di Session State if "data_transaksi" not in st.session_state: st.session_state.data_transaksi = pd.DataFrame( columns=[ "Tanggal", "Jenis", "Kategori / Sumber", "Peruntukan", "Jumlah", ] ) # 2. Form Input Transaksi (Semua Isian Manual) with st.form("form_transaksi", clear_on_submit=True): st.subheader("➕ Tambah Transaksi Baru") tanggal = st.date_input("Tanggal") jenis = st.selectbox("Jenis Transaksi", ["Pemasukan", "Pengeluaran"]) if jenis == "Pemasukan": kategori = st.text_input( "Sumber Dana", placeholder="Contoh: Dosen Pembimbing, Iuran Ex-Off, Sponsor", ) else: kategori = st.text_input( "Jenis Peruntukan", placeholder="Contoh: Cetak Proposal, Sewa Alat, Konsumsi Rapat", ) peruntukan = st.text_input( "Keterangan Detail (Opsional)", placeholder="Contoh: Rapat Koordinasi Divisi Musik", ) jumlah = st.number_input("Jumlah (Rp)", min_value=0, step=1000) submit = st.form_submit_button("Simpan Transaksi") if submit: if jumlah > 0 and kategori.strip() != "": data_baru = pd.DataFrame( { "Tanggal": [str(tanggal)], "Jenis": [jenis], "Kategori / Sumber": [kategori], "Peruntukan": [peruntukan], "Jumlah": [jumlah], } ) st.session_state.data_transaksi = pd.concat( [st.session_state.data_transaksi, data_baru], ignore_index=True ) st.success("Transaksi berhasil disimpan!") else: st.error( "Pastikan sumber/kategori diisi dan jumlah nominal lebih dari" " 0." ) st.divider() # 3. Ringkasan Saldo & Riwayat Transaksi yang Jelas di HP st.subheader("📊 Rekap & Riwayat Keuangan") df = st.session_state.data_transaksi if not df.empty: # Hitung total masuk & keluar total_masuk = df[df["Jenis"] == "Pemasukan"]["Jumlah"].sum() col1, col2 = st.columns(2) col1.metric("Total Pemasukan", f"Rp {total_masuk:,.0f}") total_keluar = df[df["Jenis"] == "Pengeluaran"]["Jumlah"].sum() col2.metric("Total Pengeluaran", f"Rp {total_keluar:,.0f}") saldo_akhir = total_masuk - total_keluar st.info(f"**Sisa Saldo Kas:** Rp {saldo_akhir:,.0f}") st.write("---") st.write("### 📝 Daftar Riwayat Detail") # Menampilkan riwayat dalam bentuk list/kartu baris ke bawah agar tidak terpotong di iPhone for index, row in df.iterrows(): # Beri warna ikon berbeda untuk pemasukan dan pengeluaran ikon = "🟢" if row["Jenis"] == "Pemasukan" else "🔴" with st.container(): st.markdown(f"**{ikon} {row['Tanggal']} — {row['Jenis']}**") st.write( f"• **Sumber/Peruntukan:** {row['Kategori / Sumber']}" ) if row["Peruntukan"]: st.write(f"• **Keterangan:** {row['Peruntukan']}") st.write( f"• **Jumlah:** **Rp {row['Jumlah']:,.0f}**" ) st.markdown("---") # 4. Tombol Ekspor ke Excel/CSV @st.cache_data def convert_df_to_excel(dataframe): return dataframe.to_csv(index=False).encode("utf-8") csv_data = convert_df_to_excel(df) st.download_button( label="📥 Download Data ke Format Spreadsheet/CSV", data=csv_data, file_name="Laporan_Keuangan_BEM.csv", mime="text/csv", ) else: st.warning("Belum ada data transaksi yang dicatat.")
